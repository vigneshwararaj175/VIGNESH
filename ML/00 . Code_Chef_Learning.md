```
# Take user input for starting number, step value, and limit
start = int(input( ))
step = int(input( ))
limit = int(input( ))

# Validate input: Ensure all values are positive integers
if start <= 0 or step <= 0 or limit <= 0:
    print("Invalid input. Please enter positive integers.")
else:
    # Initialize variables
    count = 0
    current = start

    # Generate the series using a while loop without a list
    while count < limit:
        print(current, end=", " if count < limit - 1 else "\n")  # Print with comma formatting
        current += step  # Increment by step value
        count += 1  # Increase count
```

### String Processing Loops

![[Pasted image 20260404084624.png]]

![[Pasted image 20260404140641.png]]

```
# Take user input for the transcript
transcript = input()

# Validate input: Ensure it's not empty or just spaces
if not transcript.strip():
    print("Invalid input. Please enter a valid transcript.")
else:
    # Initialize variables
    index = 0
    vowel_count = 0
    consonant_count = 0
    vowels = "aeiouAEIOU"  # Define vowels for checking

    # Process the transcript using a while loop
    while index < len(transcript):
        char = transcript[index]

        # Check if the character is a letter (ignoring spaces, numbers & special characters)
        if char.isalpha():
            if char in vowels:  # Check if it's a vowel
                vowel_count += 1
            else:  # Otherwise, it's a consonant
                consonant_count += 1
        
        index += 1  # Move to the next character

    # Print the analysis results
    print(f"Total vowels: {vowel_count}")
    print(f"Total consonants: {consonant_count}")
```
